* Claude Code IDE

#+begin_src emacs-lisp
   ;; -*- lexical-binding: t; -*-
   (use-package eat
     :commands eat)

   (use-package claude-code-ide
     :commands (claude-code-ide claude-code-ide-menu)
     :bind (("C-c C-c C-i" . claude-code-ide)
 	   ("C-c C-'" . claude-code-ide-menu))
     :config
     (setq claude-code-ide-terminal-backend 'eat)
     (setq claude-code-ide-cli-path
           (or (executable-find "claude")
               (seq-find #'file-executable-p
                         (mapcar #'expand-file-name
                                 '("~/.local/bin/claude"
                                   "/opt/homebrew/bin/claude"
                                   "/usr/local/bin/claude")))
               "claude"))
     (claude-code-ide-emacs-tools-setup))
#+end_src

* Codex IDE

#+begin_src emacs-lisp
 (use-package codex-ide
   :preface
   ;; Load the macro before package-vc generates autoloads.
   (require 'transient)
   :bind ("C-c C-;" . codex-ide-menu)
   :custom
   (codex-ide-cli-path
    (or (executable-find "codex")
        ;; GUI Emacs may not inherit the shell's PATH.
        (seq-find #'file-executable-p
                  (mapcar #'expand-file-name
                          '("~/.local/share/mise/installs/codex/latest/bin/codex"
                            "~/.local/bin/codex"
                            "/opt/homebrew/bin/codex"
                            "/usr/local/bin/codex")))
        "codex")))
#+end_src
