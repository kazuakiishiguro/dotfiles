** Wayland clipboard integration
#+begin_src emacs-lisp
(when (and (getenv "WAYLAND_DISPLAY")
           (executable-find "wl-paste"))
  (setq interprogram-paste-function
        (lambda ()
          (shell-command-to-string "wl-paste --no-newline 2>/dev/null")))
  (setq interprogram-cut-function
        (lambda (text)
          (let ((process-connection-type nil))
            (let ((proc (start-process "wl-copy" nil "wl-copy")))
              (process-send-string proc text)
              (process-send-eof proc))))))
#+end_src

** Backup and auto-save
#+begin_src emacs-lisp
(let ((backup-directory (expand-file-name "var/backups/" user-emacs-directory))
      (auto-save-directory (expand-file-name "var/auto-save/" user-emacs-directory)))
  (dolist (directory (list backup-directory auto-save-directory))
    (make-directory directory t)
    (set-file-modes directory #o700))
  (setq make-backup-files t
        backup-by-copying t
        backup-directory-alist `(("." . ,backup-directory))
        auto-save-default t
        auto-save-file-name-transforms `((".*" ,auto-save-directory t))
        auto-save-list-file-prefix (expand-file-name ".saves-" auto-save-directory)))
#+end_src

(defun my/insert-line-before ()
  "Insert a newline(s) above the line containing the cursor."
  (interactive)
  (save-excursion
    (move-beginning-of-line 1)
    (newline)))

(global-set-key (kbd "C-o") 'my/insert-line-before)

** Trim trailing whitespace
#+begin_src emacs-lisp
(use-package ws-butler
  :diminish ws-butler-mode
  :config
  ;; ws-butler trims changed lines; avoid also trimming the entire file.
  (when (bound-and-true-p my/omarchy-integration-p)
    (remove-hook 'before-save-hook #'delete-trailing-whitespace)
    ;; Theme/font hooks reload Omarchy, which otherwise re-adds that hook.
    (with-eval-after-load "omarchy"
      (remove-hook 'before-save-hook #'delete-trailing-whitespace)))
  (ws-butler-global-mode))
